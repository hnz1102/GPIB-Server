# components/ssh_bridge/CMakeLists.txt
#
# SSH-to-USB-CDC serial bridge component.
#
cmake_minimum_required(VERSION 3.16)

idf_component_register(
    SRCS         "ssh_bridge.c"
    INCLUDE_DIRS "include"
    REQUIRES     wolfssl wolfssh lwip freertos app_update
)

# Generate version_gen.h from the crate's Cargo.toml `version` field so the
# SSH banner (SSH_BRIDGE_VERSION in ssh_bridge.c) always matches it. PROJECT_DIR
# is set by embuild/cargo when invoking cmake for this component.
file(STRINGS "$ENV{PROJECT_DIR}/Cargo.toml" _ssh_bridge_cargo_version_line REGEX "^version[ \t]*=")
string(REGEX REPLACE "^version[ \t]*=[ \t]*\"([^\"]*)\".*" "\\1" SSH_BRIDGE_VERSION "${_ssh_bridge_cargo_version_line}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version_gen.h"
    "// Auto-generated by CMakeLists.txt from Cargo.toml — do not edit.\n"
    "#pragma once\n"
    "#define SSH_BRIDGE_VERSION \"${SSH_BRIDGE_VERSION}\"\n"
)
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

# wolfSSL's settings.h uses #include "FreeRTOS.h".
# In ESP-IDF 5.x FreeRTOS.h is under freertos/ subdirectory.
target_include_directories(${COMPONENT_LIB} PRIVATE
    "${IDF_PATH}/components/freertos/FreeRTOS-Kernel/include/freertos"
    "${IDF_PATH}/components/freertos/FreeRTOS-Kernel/portable/xtensa/include/freertos"
)

target_compile_options(${COMPONENT_LIB} PRIVATE
    -Wno-unused-function
    -Wno-unused-variable
)
